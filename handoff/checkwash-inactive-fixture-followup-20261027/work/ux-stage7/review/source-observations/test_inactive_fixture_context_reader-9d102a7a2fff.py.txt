"""The optional context reader preserves captured sides and bounds failed reads.

The proof function is a recorder here; it never grants credit. These controls
assert reader bytes/provenance, not a verdict that could erase valid D1 repair.
Represented repository programs are static bytes and are never executed.
"""
from collections import Counter
import datetime

import pytest

from checkwash.change import FileChange
from checkwash.config import Config
from checkwash.contract import Contract
import checkwash.engine as engine
from checkwash.report.context import ReportContext


TODAY = datetime.date(2026, 10, 27)
PRODUCER = "app/billing.py"
ARCHIVE = "app/archive.py"
PROBE = "oracle_probe/extra.py"
OLD_PRODUCER = b"def total():\n    return 78.75\n"
NEW_PRODUCER = b"def total():\n    return 0\n"
GATE = (
    "import os\nimport pytest\nfrom app.billing import total\n"
    "@pytest.fixture(autouse=True)\n"
    "def only_slow(request):\n"
    "    if request.node.get_closest_marker('slow') and not os.environ.get('SLOW'):\n"
    "        pytest.skip('off')\n"
)
UNIT = "def test_total():\n    assert total() == 78.75\n"


def recorded(monkeypatch, support=(), *, queries=(PRODUCER,), head=None, fault=None):
    before = ("# reader-phase-base\n" + GATE + UNIT).encode()
    after = ("# reader-phase-head\n" + GATE + UNIT).encode()
    old_after = (GATE + "\n").encode()
    snapshot = {"app/__init__.py": b"", PRODUCER: OLD_PRODUCER,
                "tests/test_old.py": old_after, "tests/test_new.py": after,
                **(head or {})}
    root_reads = Counter()
    events = []
    context = ReportContext()

    def root_read(path):
        root_reads[path] += 1
        if path == PROBE and fault is not None:
            if fault == "oversize":
                return b"#" * 65537
            if fault == "invalid-type":
                return "not source bytes"
            if fault == "io-error":
                raise OSError("owned optional-reader control")
        return snapshot.get(path)

    def capture(source, units, path, language, *, source_reader=None, **_kwargs):
        for _name, side in units:
            side.oracle_context = ""
            side.oracle_assertion_contexts = {}
        if language != "python" or not any(name == "test_total" for name, _side in units):
            return
        assert callable(source_reader)
        side = 0 if b"reader-phase-base" in source else 1
        for query in queries:
            try:
                data = source_reader(query)
            except (ValueError, OSError) as error:
                events.append((side, query, type(error).__name__, None))
            else:
                events.append((side, query, data, context._sources.get((query, side))))

    monkeypatch.setattr(engine, "mark_contexts", capture)
    engine.analyze(
        [FileChange("tests/test_old.py", "modified", before, old_after),
         FileChange("tests/test_new.py", "added", None, after), *support],
        Config(), Contract(), [], TODAY,
        root_reader=root_read, head_reader=snapshot.get,
        root_searcher=lambda _needles: [], report_context=context,
    )
    # File traversal is deliberately irrelevant to side correctness.
    return sorted(events, key=lambda event: (event[0], event[1])), root_reads


@pytest.mark.parametrize("reversed_records", [False, True])
@pytest.mark.parametrize("backslash", [False, True])
def test_rename_source_resurrection_keeps_both_captured_sides(monkeypatch, reversed_records, backslash):
    old = PRODUCER.replace("/", "\\") if backslash else PRODUCER
    records = [FileChange(ARCHIVE, "modified", OLD_PRODUCER, OLD_PRODUCER, old_path=old),
               FileChange(PRODUCER, "added", None, NEW_PRODUCER)]
    if reversed_records:
        records.reverse()
    events, _reads = recorded(monkeypatch, records,
                              head={PRODUCER: NEW_PRODUCER, ARCHIVE: OLD_PRODUCER})
    assert events == [(0, PRODUCER, OLD_PRODUCER, OLD_PRODUCER.decode()),
                      (1, PRODUCER, NEW_PRODUCER, NEW_PRODUCER.decode())]


@pytest.mark.parametrize("reversed_records", [False, True])
def test_conflicting_actual_before_bytes_withhold_only_optional_read(monkeypatch, reversed_records):
    records = [FileChange(ARCHIVE, "modified", OLD_PRODUCER, OLD_PRODUCER, old_path=PRODUCER),
               FileChange(PRODUCER, "modified", b"def total():\n    return 9\n", NEW_PRODUCER)]
    if reversed_records:
        records.reverse()
    events, _reads = recorded(monkeypatch, records,
                              head={PRODUCER: NEW_PRODUCER, ARCHIVE: OLD_PRODUCER})
    assert events[0][:3] == (0, PRODUCER, "ValueError")
    assert events[1][:3] == (1, PRODUCER, NEW_PRODUCER)


@pytest.mark.parametrize("reversed_records", [False, True])
def test_explicit_destination_absence_cannot_be_erased_by_rename(monkeypatch, reversed_records):
    records = [FileChange(ARCHIVE, "modified", OLD_PRODUCER, OLD_PRODUCER, old_path=PRODUCER),
               FileChange(ARCHIVE, "deleted", b"def stale():\n    return 1\n", None)]
    if reversed_records:
        records.reverse()
    events, _reads = recorded(monkeypatch, records, queries=(ARCHIVE,), head={ARCHIVE: OLD_PRODUCER})
    assert events[1][:3] == (1, ARCHIVE, "ValueError")


@pytest.mark.parametrize("fault", ["oversize", "invalid-type", "io-error"])
def test_refused_unchanged_source_is_negative_cached_across_sides(monkeypatch, fault):
    events, reads = recorded(monkeypatch, queries=(PROBE, PROBE), fault=fault)
    assert len(events) == 4 and all(event[2] in {"ValueError", "OSError"} for event in events)
    assert reads[PROBE] == 1


def test_unchanged_valid_bytes_are_read_once_and_reported_on_both_sides(monkeypatch):
    raw = b"# owned context\n"
    events, reads = recorded(monkeypatch, queries=(PROBE, PROBE), head={PROBE: raw})
    assert reads[PROBE] == 1
    assert events == [(side, PROBE, raw, raw.decode()) for side in (0, 1) for _repeat in range(2)]
