def ordinary_check(raw):
    tree = ast.parse(raw)
    assertions = [ast.dump(node) for node in ast.walk(tree) if isinstance(node, ast.Assert)]
    before_assertions = [ast.dump(node) for node in ast.walk(original_tree) if isinstance(node, ast.Assert)]
    assert assertions == before_assertions
    normalized = copy.deepcopy(tree)
    added_imports = [node for node in normalized.body if isinstance(node, ast.ImportFrom)
                     and node.module == "checkwash.gitio.snapshot"]
    assert len(added_imports) == 1
    assert [(alias.name, alias.asname) for alias in added_imports[0].names] == [("search_source_mapping", None)]
    normalized.body.remove(added_imports[0])
    fn = next(node for node in normalized.body if isinstance(node, ast.FunctionDef)
              and node.name == "test_a_test_moved_under_the_gate_keeps_its_move_credit")
    calls = [node for node in ast.walk(fn) if isinstance(node, ast.Call)
             and isinstance(node.func, ast.Name) and node.func.id == "analyze"]
    assert len(calls) == 1
    added_keywords = [keyword.arg for keyword in calls[0].keywords
                      if keyword.arg in {"root_searcher", "root_path_lister", "root_batch_reader"}]
    calls[0].keywords = [keyword for keyword in calls[0].keywords if keyword.arg not in added_keywords]
    assert ast.dump(normalized) == ast.dump(original_tree)
    return {"all_assertion_asts_identical_to_fbc": True, "assertion_count": len(assertions),
            "entire_module_ast_identical_after_only_added_import_and_context_keywords_removed": True,
            "added_context_keywords": added_keywords}

runs = []
history = {}
expected = {1: (274, 0, 0), 2: (179, 143, 1), 3: (322, 1, 1), 4: (323, 0, 0)}
for number in range(1, 5):

    assert receipt["validation_kind"] == "WORKING_TREE_DEVELOPMENT_NOT_COMMIT_QUALIFICATION"

    copies = {}
    for row in manifest["files"]:
        raw = read(historical / (row["path"] + ".txt"))
        assert len(raw) == row["bytes"] and sha(raw) == row["sha256"] == inputs[row["path"]]
        copies[row["path"]] = raw
    history[number] = copies

    assert not removed and different <= owned

    # byte-identical to the fixed fbc seal. New paths may only be owned six.
    for path, digest in sealed.items():
        if path not in owned:
            assert inputs[path] == digest
    xml_raw = read(run / "junit.xml")
    xml = ET.fromstring(xml_raw)
    suites = list(xml.iter("testsuite"))
    cases = list(xml.iter("testcase"))

    errors = [case for case in cases if case.find("error") is not None]

    assert (passed, len(failed), len(errors), len(skipped), receipt["exit_code"]) == (wanted_passed, wanted_failed, 0, 0, wanted_exit)
    assert sum(int(suite.attrib["tests"]) for suite in suites) == len(cases)
    assert sum(int(suite.attrib["failures"]) for suite in suites) == len(failed)
    assert sum(int(suite.attrib["errors"]) for suite in suites) == len(errors)
    assert sum(int(suite.attrib["skipped"]) for suite in suites) == len(skipped)
