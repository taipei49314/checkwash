"""Closed inactive-fixture proofs must preserve the actual oracle obligation.

All subject programs below are source bytes for static analysis, never executed.
The ordinary #358 expectation is retained in its original test file. These
controls exercise the same move through complete, separate BASE/HEAD sources.
"""
import datetime

import pytest

from checkwash.change import EngineError, FileChange
from checkwash.config import Config
from checkwash.contract import Contract
from checkwash.engine import analyze


TODAY = datetime.date(2026, 10, 27)
OLD = "tests/test_billing.py"
NEW = "tests/test_other.py"
PRODUCER = "app/billing.py"
APP = {"app/__init__.py": b"", PRODUCER: b"def total():\n    return 78.75\n"}
GATE = (
    "@pytest.fixture(autouse=True)\n"
    "def only_slow(request):\n"
    "    if request.node.get_closest_marker('slow') and not os.environ.get('SLOW'):\n"
    "        pytest.skip('slow tests are off')\n\n"
)
UNIT = "def test_total():\n    assert total() == 78.75\n"
MOVE_CREDIT = {"ASSERTION_MOVED", "RESTRUCTURED", "SPLIT_OR_RENAMED", "DUPLICATE_REMAINS"}


def module(tests=UNIT, *, gate=GATE, prelude=""):
    return "import os\nimport pytest\nfrom app.billing import total\n\n" + prelude + "\n" + gate + tests


def moved(*, before_tests=UNIT, after_tests=None, gate=GATE, before_gate=None,
          prelude="", before_prelude=None, files=None, changed_files=(),
          root=True, root_override=None, head_override=None):
    """A retained fixture, a removed real test and its proposed live arrival.

    Every changed support source carries its actual two sides in FileChange;
    strict readers supply only unchanged HEAD sources. No HEAD-only snapshot
    is presented as BASE authority for a changed producer or initializer.
    """
    if after_tests is None:
        after_tests = before_tests
    before = module(before_tests, gate=gate if before_gate is None else before_gate,
                    prelude=prelude if before_prelude is None else before_prelude)
    old_after = module("", gate=gate, prelude=prelude)
    arrival = module(after_tests, gate=gate, prelude=prelude)
    changes = [FileChange(OLD, "modified", before.encode(), old_after.encode()),
               FileChange(NEW, "added", None, arrival.encode()), *changed_files]
    snapshot = {**APP, OLD: old_after.encode(), NEW: arrival.encode(), **(files or {})}
    for change in changed_files:
        if change.after is None:
            snapshot.pop(change.path, None)
        else:
            snapshot[change.path] = change.after
    return analyze(changes, Config(), Contract(), [], TODAY,
                   head_reader=head_override or snapshot.get,
                   root_reader=(root_override or snapshot.get) if root else None,
                   root_searcher=lambda _query: [])


def assert_blocked(result, *, unit="test_total"):
    _ir, findings, verdict = result
    disabled = [finding for finding in findings
                if finding.rule == "TEST_DISABLED" and finding.path == OLD and finding.unit == unit]
    assert verdict == "block"
    assert disabled and all(finding.severity == "high" for finding in disabled)
    assert all(not MOVE_CREDIT.intersection(finding.deescalators) for finding in disabled)


def test_closed_unmarked_move_preserves_the_original_358_expectation():
    _ir, findings, verdict = moved()
    assert verdict == "pass"
    disabled = [finding for finding in findings if finding.rule == "TEST_DISABLED"]
    assert disabled and all(finding.severity == "info" for finding in disabled)
    assert all("ASSERTION_MOVED" in finding.deescalators for finding in disabled)


def test_one_inactive_fixture_can_certify_each_unmarked_unit_separately():
    units = UNIT + "\ndef test_second():\n    assert total() == 78.75\n"
    _ir, findings, verdict = moved(before_tests=units)
    assert verdict == "pass"
    disabled = [finding for finding in findings if finding.rule == "TEST_DISABLED"]
    assert {finding.unit for finding in disabled} == {"test_total", "test_second"}
    assert all(finding.severity == "info" and "ASSERTION_MOVED" in finding.deescalators
               for finding in disabled)


def test_a_live_sibling_cannot_lend_its_certificate_to_a_newly_slow_unit():
    units = UNIT + "\ndef test_second():\n    assert total() == 78.75\n"
    arrival = UNIT + "\n@pytest.mark.slow\ndef test_second():\n    assert total() == 78.75\n"
    result = moved(before_tests=units, after_tests=arrival)
    assert_blocked(result, unit="test_second")
    assert any(finding.unit == "test_total" and finding.severity == "info"
               and "ASSERTION_MOVED" in finding.deescalators
               for finding in result[1] if finding.rule == "TEST_DISABLED")

@pytest.mark.parametrize("after", [
    b"def total():\n    return 0\n",
    b"def total():\n    return 78.75 + 0\n",
])
def test_two_pure_producer_sources_are_not_the_same_source_certificate(after):
    change = FileChange(PRODUCER, "modified", APP[PRODUCER], after)
    ir, findings, _verdict = moved(changed_files=[change])
    disabled = [finding for finding in findings
                if finding.rule == "TEST_DISABLED" and finding.path == OLD and finding.unit == "test_total"]
    assert disabled
    assert all(not MOVE_CREDIT.intersection(finding.deescalators) for finding in disabled)
    before = next(unit.before for file in ir.files if file.path == OLD
                  for unit in file.units if unit.qualname == "test_total" and unit.before is not None)
    arrival = next(unit.after for file in ir.files if file.path == NEW
                   for unit in file.units if unit.qualname == "test_total" and unit.after is not None)
    assert not before.oracle_context or not arrival.oracle_context or before.oracle_context != arrival.oracle_context
    # D1 repair evidence for the changed production source is independent
    # of D2 relocation proof; its existing verdict/de-escalation is retained.

@pytest.mark.parametrize("failure", ["exception", "nonbytes", "missing-producer", "oversize-producer"])
def test_untrusted_or_bounded_out_source_reading_never_produces_move_credit(failure):
    def read(path):
        if path == PRODUCER:
            if failure == "exception":
                raise EngineError("synthetic unavailable strict source")
            if failure == "nonbytes":
                return "def total(): return 78.75\n"
            if failure == "missing-producer":
                return None
            return APP[PRODUCER] + b"#" * 65_537
        return APP.get(path)
    try:
        result = moved(root_override=read, head_override=read)
    except EngineError:
        # The public strict-reader refusal is also a fail-closed result;
        # it must not become successful relocation evidence.
        return
    assert_blocked(result)
