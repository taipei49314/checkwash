"""Plan/prepare one frozen diagnostic supplement; publish only a reviewed digest.

No product/fixture/test is imported or run. Signed stage5 metadata controllers
provide the unchanged privacy and Git/source guards. No qualification is rerun.
"""
from __future__ import annotations
import argparse
from datetime import datetime, timezone
import hashlib
import io
import json
import os
from pathlib import Path
import re
import stat
import sys
import zipfile

HERE = Path(__file__).resolve().parent
SESSION = HERE.parents[2]
STAGE = SESSION / "work/ux-stage6"
OUT = SESSION / "outputs/checkwash-diagnostics-20261027"
SOURCE = "fbc9374dc3bcbfc47f403aec0c36e6b51dd7b86c"
PARENT = "ec4b80ac162d1fbdb73c329ee39d87992d4fbafe"
BRANCH = "handoff/core-010-pju-20261027-8a8941dc"
PREFIX = "handoff/checkwash-diagnostics-20261027"
PURPOSE = "DIAGNOSTIC_SUPPLEMENT_NOT_ACCEPTANCE_NOT_RELEASE"
HELPERS = {
    "privacy": ("work/ux-stage5/checkpoint-pack/pack_checkpoint.py",
                "ab1ab82cb38f8271a7b8bcbded31f35cc7f43e752a1e5ed7c8c0b9aba2a08928"),
    "publisher": ("work/ux-stage5/checkpoint-pack/publish_checkpoint.py",
                  "9b559a07f3e46fdb2b4bf8614a06c9aa135d7bb5aafb37b60bba9f6a4be4b7be"),
}
EXCLUDED = {"checkpoint-pack", ".git", ".venv", "venv", "site-packages",
            "__pycache__", ".pytest_cache", ".ruff_cache", ".mypy_cache",
            "node_modules", "tmp", "temp", "owned-tmp", "pytest-temp", "runtime"}
EXPORT_SUFFIXES = {".md", ".json", ".patch", ".txt"}
MAX_FILE = 16 * 1024 * 1024
MAX_TOTAL = 64 * 1024 * 1024


def require(value, message):
    if not value:
        raise ValueError(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def encoded(value):
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode("utf-8")


def read_regular(path):
    path = Path(os.path.abspath(path))
    require(path.is_relative_to(SESSION), "input outside this task")
    current = SESSION
    for part in (None,) + path.relative_to(SESSION).parts:
        if part is not None:
            current /= part
        info = current.lstat()
        require(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 1024,
                "link/reparse input refused")
        require(stat.S_ISDIR(info.st_mode) if current != path else stat.S_ISREG(info.st_mode),
                "nonregular input refused")
    before = path.stat()
    require(before.st_size <= MAX_FILE, "diagnostic input exceeds16MiB")
    raw = path.read_bytes()
    after = path.stat()
    require((before.st_size, before.st_mtime_ns, before.st_ino) ==
            (after.st_size, after.st_mtime_ns, after.st_ino) and len(raw) == before.st_size,
            "diagnostic input changed during read")
    return raw


def helpers():
    result = {}
    for kind, (name, expected) in HELPERS.items():
        path = SESSION / name
        raw = read_regular(path)
        require(sha(raw) == expected, "reviewed metadata helper changed: " + kind)
        # Execute exactly the already reviewed metadata bytes, not a later reread.
        # The __main__ CLI guard stays false; these helpers import only stdlib.
        namespace = {"__name__": "signed_diagnostic_metadata_" + kind, "__file__": str(path)}
        exec(compile(raw, str(path), "exec"), namespace)
        result[kind] = namespace
    return result


def content_guard(raw, name, privacy, denied, synthetic):
    raw.decode("utf-8")
    require(b"\0" not in raw and not zipfile.is_zipfile(io.BytesIO(raw)), "binary/archive diagnostic refused")
    # No allow_reviewed_header_literals argument: the default remains false.
    reason = privacy["inspect"](raw, name, denied, synthetic)
    require(reason is None, "diagnostic privacy guard refused " + name + ": " + str(reason))


def directory_chain(path, create=False):
    path = Path(os.path.abspath(path))
    require(path.is_relative_to(SESSION), "directory outside this task")
    current = SESSION
    for part in (None,) + path.relative_to(SESSION).parts:
        if part is not None:
            current /= part
        try:
            info = current.lstat()
        except FileNotFoundError:
            require(create, "required directory is missing")
            current.mkdir()
            info = current.lstat()
        require(stat.S_ISDIR(info.st_mode) and not stat.S_ISLNK(info.st_mode)
                and not getattr(info, "st_file_attributes", 0) & 1024,
                "directory link/reparse refused before writing")


def payload(selection_path, pub, privacy):
    selection_raw = read_regular(selection_path)
    selection = json.loads(selection_raw)
    require(set(selection) == {"source_sha", "purpose", "root_frozen", "files"}, "unexpected selection schema")
    require(selection["source_sha"] == SOURCE and selection["purpose"] == PURPOSE
            and selection["root_frozen"] is True, "root's fixed-source frozen selection required")
    require(isinstance(selection["files"], list) and bool(selection["files"]), "empty selection")
    seal = json.loads(read_regular(SESSION / "work/ux-stage5/seal/source-seal.json"))
    denied, synthetic = privacy["guards"]({r["path"]: r["sha256"] for r in seal["files"]})
    files, inputs = {}, []
    for row in selection["files"]:
        require(set(row) == {"source", "export", "bytes", "sha256"}, "unexpected selected-row schema")
        source = pub["logical"](row["source"])
        target = pub["logical"](row["export"])
        require(source.startswith("work/ux-stage6/") and target in {source, source + ".txt"},
                "selection must preserve stage6 task layout")
        parts = Path(source).parts
        require(not any(p.lower() in EXCLUDED or p.lower().endswith("-runtime") for p in parts),
                "runtime/cache/temp/controller source refused")
        require(Path(target).suffix.lower() in EXPORT_SUFFIXES, "unsupported diagnostic export suffix")
        require(not source.lower().endswith((".zip", ".whl", ".pyz", ".tar", ".gz", ".exe", ".dll", ".pyd")),
                "archive/runtime source refused")
        if source.lower().endswith(".py"):
            require(target == source + ".txt", "Python source must be exported as .py.txt")
        raw = read_regular(SESSION / source)
        require(type(row["bytes"]) is int and len(raw) == row["bytes"]
                and re.fullmatch(r"[0-9a-f]{64}", row["sha256"])
                and sha(raw) == row["sha256"], "frozen diagnostic bytes differ: " + source)
        content_guard(raw, source, privacy, denied, synthetic)
        require(target not in files, "duplicate diagnostic export")
        files[target] = raw
        inputs.append(row)
    for original, target in ((HERE / "diagnostic_checkpoint.py", "TOOLS/diagnostic_checkpoint.py.txt"),
                             (HERE / "README.md", "TOOLS/README.md")):
        raw = read_regular(original)
        content_guard(raw, target, privacy, denied, synthetic)
        files[target] = raw
    files["SELECTION.json"] = selection_raw
    files["checkpoint.json"] = encoded({
        "source_sha": SOURCE, "purpose": PURPOSE, "kind": "FROZEN_DIAGNOSTIC_SUPPLEMENT",
        "acceptance_state": "NONPASS_NOT_REASSESSED", "previous_complete_backup": PARENT,
        "product_publication_performed": False, "version_1_0_requires_human_decision": True,
        "product_fixture_test_model_calls_by_backup_tool": 0,
        "note": "Copied diagnostic outcomes/proposals keep their recorded state; no all-tests-PASS assertion is made.",
    })
    files["00-START-HERE.md"] = (
        "# CheckWash diagnostic supplement\n\nBackup baseline: `" + SOURCE + "`. Purpose: `" + PURPOSE + "`.\n\n"
        "Each included receipt retains its own actual source. See [work/ux-stage6/00-CONTINUE.md]"
        "(work/ux-stage6/00-CONTINUE.md) for the separate15ac consumer candidate.\n"
        "That candidate does not inherit the fbc full-suite or qualification receipts.\n\n"
        "These are frozen stage6 observations/proposals, not new product acceptance or a release.\n"
        "Each source file retains its own observed / inert / NOT_EXECUTED status.\n"
        "The preceding complete backup is `" + PARENT + "`; its package and results stay unchanged.\n"
        "The handoff tree inherits its parent's product root; restore the separate exact fbc source,\n"
        "not the handoff metadata commit, when continuing product work.\n\n"
        "MANIFEST.json maps original task-relative inputs to exported bytes. Python sources end in\n"
        ".py.txt for transport and are not automatically executed. This supplement contains no ZIP/runtime.\n"
        "Tools require the exact stage5 metadata helpers restored from the preceding fbc evidence bundle.\n"
        "No qualification is rerun, no historical/model batch is admitted, and Nelson decides before1.0.\n"
    ).encode("utf-8")
    require(len({name.casefold() for name in files}) == len(files), "casefold export collision")
    rows = [{"path": name, "bytes": len(raw), "sha256": sha(raw)} for name, raw in sorted(files.items())]
    manifest = {"source_sha": SOURCE, "purpose": PURPOSE, "selection_sha256": sha(selection_raw),
                "original_inputs": inputs, "files": rows, "manifest_excludes_itself": True,
                "metadata_helper_pins": HELPERS, "producer_sha256": sha(read_regular(Path(__file__))),
                "product_fixture_test_model_calls_by_tool": 0, "new_acceptance_claim": False}
    files["MANIFEST.json"] = encoded(manifest)
    for name, raw in files.items():
        content_guard(raw, name, privacy, denied, synthetic)
    require(sum(map(len, files.values())) <= MAX_TOTAL, "diagnostic payload exceeds64MiB")
    return files, sha(selection_raw)


def disk_payload(pub):
    require(OUT.exists(), "prepare the fixed output first")
    directory_chain(OUT)
    files = {}
    for folder, dirs, names in os.walk(OUT, followlinks=False):
        for name in dirs:
            info = (Path(folder) / name).lstat()
            require(stat.S_ISDIR(info.st_mode) and not stat.S_ISLNK(info.st_mode)
                    and not getattr(info, "st_file_attributes", 0) & 1024, "output link/reparse directory")
        for name in names:
            path = Path(folder) / name
            relative = pub["logical"](path.relative_to(OUT).as_posix())
            files[relative] = read_regular(path)
    return files


def prepare(files):
    # Fresh files only. Existing identical bytes are harmless; no replacement.
    directory_chain(OUT, create=True)
    for name, raw in sorted(files.items()):
        target = OUT / name
        directory_chain(target.parent, create=True)
        if target.exists():
            require(read_regular(target) == raw, "existing diagnostic payload differs")
            continue
        with target.open("xb") as stream:
            stream.write(raw)
        require(read_regular(target) == raw, "prepared diagnostic bytes differ")


def parent_guard(git):
    allowed = {"https://github.com/taipei49314/checkwash.git", "git@github.com:taipei49314/checkwash.git",
               "ssh://git@github.com/taipei49314/checkwash.git"}
    require(git.run("remote", "get-url", "origin").decode().strip() in allowed, "unexpected fetch origin")
    pushes = git.run("remote", "get-url", "--push", "--all", "origin").decode().splitlines()
    require(len(pushes) == 1 and pushes[0] in allowed, "unexpected/multiple effective push origins")
    tip = git.run("ls-remote", "--heads", "origin", "refs/heads/" + BRANCH).decode().split()
    require(tip == [PARENT, "refs/heads/" + BRANCH], "handoff parent moved; reconcile instead of retrying")
    require(git.run("cat-file", "-t", PARENT).strip() == b"commit", "exact ec4 parent missing")
    require(not git.run("ls-tree", "-r", "--name-only", PARENT, "--", PREFIX), "diagnostic prefix already exists")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--selection", type=Path, required=True)
    parser.add_argument("--label", required=True)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument("--prepare", action="store_true", help="Write only the new diagnostic output payload")
    mode.add_argument("--publish", action="store_true", help="Requires explicit reviewed plan digest")
    parser.add_argument("--expect-plan-sha256")
    args = parser.parse_args()
    require(re.fullmatch(r"[a-z0-9][a-z0-9-]{0,47}", args.label), "invalid label")
    require(not args.publish or re.fullmatch(r"[0-9a-f]{64}", args.expect_plan_sha256 or ""),
            "publish requires the reviewed digest")
    helper = helpers()
    pub, privacy = helper["publisher"], helper["privacy"]
    stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S.%fZ")
    run = HERE / "runs" / (stamp + "-" + args.label)
    run.mkdir(parents=True, exist_ok=False)
    (run / "no-hooks").mkdir()
    git = pub["Git"](run)
    before, index_before = pub["source_guard"](git)
    parent_guard(git)
    files, selection_sha = payload(args.selection, pub, privacy)
    if args.prepare:
        prepare(files)
    if args.prepare or args.publish:
        require(disk_payload(pub) == files, "output payload is not the exact frozen selection")
    rows = [{"path": n, "bytes": len(b), "sha256": sha(b)} for n, b in sorted(files.items())]
    plan = {"source_sha": SOURCE, "purpose": PURPOSE, "parent": PARENT, "branch": BRANCH,
            "prefix": PREFIX, "files": rows, "bytes": sum(map(len, files.values())),
            "selection_sha256": selection_sha, "source_guard": before,
            "tool_sha256": sha(read_regular(Path(__file__))), "helper_pins": HELPERS,
            "acceptance_state": "NONPASS_NOT_REASSESSED", "publication_performed": False}
    plan_sha = sha(encoded(plan))
    (run / "plan.json").write_bytes(encoded(plan))
    (run / "plan.sha256").write_text(plan_sha + "\n", encoding="ascii")
    again, index_again = pub["source_guard"](git)
    require(again == before and index_again == index_before
            and payload(args.selection, pub, privacy) == (files, selection_sha), "source/inputs changed during plan")
    if not args.publish:
        print(json.dumps({"mode": "PREPARED_NO_GIT_WRITES" if args.prepare else "PLAN_ONLY_NO_GIT_WRITES",
                          "plan": str(run / "plan.json"), "plan_sha256": plan_sha,
                          "files": len(files), "bytes": plan["bytes"]}, indent=2))
        return
    require(args.expect_plan_sha256 == plan_sha, "reviewed plan changed; do not publish")
    alternate = run / "alternate.index"
    git.write_enabled = True
    git.run("read-tree", PARENT, alternate=alternate, writes=True)
    staged = []
    for row in rows:
        oid = git.run("hash-object", "-w", "--stdin", data=files[row["path"]],
                      alternate=alternate, writes=True).decode().strip()
        target = PREFIX + "/" + row["path"]
        git.run("update-index", "--add", "--cacheinfo", "100644", oid, target,
                alternate=alternate, writes=True)
        staged.append({**row, "path": target, "git_blob": oid})
    tree = git.run("write-tree", alternate=alternate, writes=True).decode().strip()
    commit = git.run("commit-tree", tree, "-p", PARENT, data=("checkpoint: frozen stage6 diagnostics\n\n" + PURPOSE + "\n").encode(),
                     alternate=alternate, writes=True).decode().strip()
    changes = [x.decode() for x in git.run("diff-tree", "--no-commit-id", "--no-renames", "--name-status", "-r", "-z", PARENT, commit).split(b"\0") if x]
    require(len(changes) == 2 * len(files), "unexpected prepared diagnostic delta")
    require(all(status == "A" and name.startswith(PREFIX + "/")
                for status, name in zip(changes[::2], changes[1::2])), "delta outside new prefix or deletion/modification")
    prepared = {**plan, "plan_sha256": plan_sha, "checkpoint_commit": commit, "tree": tree,
                "files": staged, "remote_ref_verified": False}
    (run / "prepared-commit.json").write_bytes(encoded(prepared))
    prepush, prepush_index = pub["source_guard"](git)
    require(prepush == before and prepush_index == index_before
            and payload(args.selection, pub, privacy) == (files, selection_sha)
            and disk_payload(pub) == files, "source/input/output changed before push")
    parent_guard(git)
    git.run("push", "--porcelain", "--no-follow-tags", "--recurse-submodules=no", "origin",
            commit + ":refs/heads/" + BRANCH, writes=True, timeout=180)
    tip = git.run("ls-remote", "--heads", "origin", "refs/heads/" + BRANCH).decode().split()
    require(tip == [commit, "refs/heads/" + BRANCH], "unknown push outcome; preserve prepared receipt and reconcile")
    final, final_index = pub["source_guard"](git)
    require(final == before and final_index == index_before
            and payload(args.selection, pub, privacy) == (files, selection_sha)
            and disk_payload(pub) == files, "source/input/output changed after push")
    result = {**prepared, "remote_ref_verified": True, "product_index_unchanged": True,
              "product_source_bytes_unchanged": True, "local_main_tags_unchanged": True,
              "product_fixture_test_model_calls_by_tool": 0, "publication_performed": False}
    (run / "push-receipt.json").write_bytes(encoded(result))
    (HERE / "remote-state.json").write_bytes(encoded(result))
    print(json.dumps({"checkpoint_commit": commit, "source_sha": SOURCE, "purpose": PURPOSE,
                      "remote_ref_verified": True, "files": len(staged)}, indent=2))


if __name__ == "__main__":
    try:
        main()
    except Exception as error:
        print("Diagnostic supplement stopped: " + type(error).__name__ + ": " + str(error), file=sys.stderr)
        raise SystemExit(1)
