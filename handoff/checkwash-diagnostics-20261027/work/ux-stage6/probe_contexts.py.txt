def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def observe(label, changes, when, snapshot=None):
    lookups, marks = [], []
    original = engine.mark_contexts

    def marked(source, units, path, language, **kwargs):
        returned = original(source, units, path, language, **kwargs)
        marks.append({'path': path, 'language': language,
                      'source_sha256': sha(source) if source is not None else None,
                      'startup_known': kwargs.get('startup_known'),
                      'startup': kwargs.get('startup'),
                      'needed': sorted(kwargs.get('needed') or []),
                      'units': [{'name': name, 'span': side.span,
                                 'oracle_context': side.oracle_context,
                                 'oracle_assertion_contexts': side.oracle_assertion_contexts.copy()}
                                for name, side in units]})
        return returned

    def reader(channel):
        def read(path):
            value = snapshot.get(path)
            lookups.append({'channel': channel, 'path': path,
                            'sha256': sha(value) if value is not None else None})
            return value
        return read

    kwargs = {} if snapshot is None else {'head_reader': reader('HEAD'), 'root_reader': reader('ROOT')}
    row = {'label': label, 'date': str(when), 'closed_snapshot': snapshot is not None,
           'changes': [{'path': c.path, 'status': c.status,
                        'before_sha256': sha(c.before) if c.before is not None else None,
                        'after_sha256': sha(c.after) if c.after is not None else None,
                        'before_utf8': c.before.decode() if c.before is not None else None,
                        'after_utf8': c.after.decode() if c.after is not None else None}
                       for c in changes],
           'snapshot': {p: {'sha256': sha(b), 'utf8': b.decode()} for p, b in (snapshot or {}).items()}}
    engine.mark_contexts = marked
    try:
        ir, findings, verdict = engine.analyze(changes, Config(), Contract(), [], when, **kwargs)
        row.update({'verdict': verdict, 'findings': [asdict(f) for f in findings],
                    'files': [{'path': f.path, 'role': f.role, 'language': f.language,
                               'units': [{'qualname': u.qualname,
                                          'before': asdict(u.before) if u.before else None,
                                          'after': asdict(u.after) if u.after else None,
                                          'after_live': engine.unit_is_live(u.after, f.constants) if u.after else None}
                                         for u in f.units]} for f in ir.files]})
    finally:
        engine.mark_contexts = original
    row['startup_lookups'] = lookups
    row['mark_context_calls'] = marks
    results.append(row)
    (OUT / (label + '.json')).write_text(json.dumps(row, sort_keys=True, indent=2) + '\n', encoding='utf-8')
